import csv, math, resource, sys
import numpy as np
resource.setrlimit(resource.RLIMIT_DATA, (1024*2**20, 1024*2**20))
D = sys.argv[1] if len(sys.argv) > 1 else '.'
CPI25 = float(sys.argv[2]) if len(sys.argv) > 2 else 321.943
def load(fn):

fw = load('draws-fusion-first-watt.csv'); ir = load('draws-irena-lcoe.csv')

assert all(int(a['draw']) == i == int(b['draw']) for i, (a, b) in enumerate(zip(fw, ir)))

succ = np.array([float(r['successor_year']) for r in fw]); start = np.array([float(r['start']) for r in fw])

R0 = 0.075651                         # driver r_offshore p50 (refit 2026-10-16)
ASOF = 2026 + (289 - 0.5) / 365       # core/drivers.py decimal_year(2026-10-16) = 2026.7904
def start_of(route, slip, succ, arc=2032.5):
    return np.where(route == 'helion', np.maximum(2028, ASOF + (2028 - ASOF) * slip),
           np.where(route == 'arc', np.maximum(2030, ASOF + (arc - ASOF) * slip),
           np.where(route == 'successor', succ, np.inf)))
chk = start_of(route, slip, succ)
assert np.allclose(np.where(np.isinf(chk), 0, chk), np.where(np.isinf(start), 0, start), atol=1e-5)
g = np.random.default_rng(20261016)
Z_C = g.standard_normal(N); U_nl = g.random(N); U_re = g.random(N); U_mix = g.random(N)
CPI24, CPI21 = 313.689, 270.970
BASE = dict(C0_med=170., C0_sd=.25, p_nl=.20, thr=100. * CPI24 / CPI25, r_fixed=None, follow=0.0)
def gp(st, **kw):
    p = dict(BASE); p.update(kw)
    C0 = np.exp(np.log(p['C0_med']) + p['C0_sd'] * Z_C)
    r = r_off if p['r_fixed'] is None else r_off * p['r_fixed'] / R0      # same draw, median rescaled
    if p.get('r_sd') is not None:
        r = np.exp(np.log(R0) + (np.log(r_off) - np.log(R0)) * p['r_sd'] / .55)
    L = np.log(C0 / p['thr']); cheap = L <= 0
    dt = np.where(cheap, 0., L / r + p['follow'])
    return np.where(np.isinf(st), np.inf, np.where(cheap, st, np.where(U_nl < p['p_nl'], np.inf, st + dt)))
def summ(x):
    s = np.sort(x); P = float(np.mean(x <= 2100))
    q = lambda pr: (None if s[int(math.ceil(N * pr)) - 1] > 2100 else float(s[int(math.ceil(N * pr)) - 1]))
    ok = x[x <= 2100]
    return P, q(.1), q(.5), q(.9), (float(np.median(ok)) if len(ok) else None)
def fmt(x):
    P, a, b, c, m = summ(x)
    f = lambda v: 'null' if v is None else f'{v:.2f}({math.floor(v)})'
    return f'P {P:.4f} p10 {f(a)} p50 {f(b)} p90 {f(c)} cond_med {f(m)}'
def reassign(src, dst, frac, st_route):
    rt = st_route.copy(); sel = (rt == src) & (U_re < frac); rt[sel] = dst; return rt

    xs=[x for x,_,_ in p]; ys=[y for _,y,_ in p]; mx=st.mean(xs); my=st.mean(ys)
