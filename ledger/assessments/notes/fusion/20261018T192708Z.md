# KPI window statistics, fusion, as of 2026-10-18 (run 20261018T192708Z). Same method as prev (20261013T134057Z:deep).
import subprocess, csv, io, statistics as st, math
out = subprocess.run(['uv','run','/home/pasky/projects/hai/grand-endeavors-agent-worktree-0001/core/ledger.py','series','fusion','elec-price-household-world-popw-real','--as-of','2026-10-18'],
                     capture_output=True, text=True, check=True).stdout
rows = [r for r in csv.reader(io.StringIO(out)) if r and r[0][:4].isdigit()]
pts = [(int(o[:4]) + (int(o[-1]) - 0.5) / 4, float(v), o) for o, v, *_ in rows]
def ols(p):
    xs=[x for x,_,_ in p]; ys=[y for _,y,_ in p]; mx=st.mean(xs); my=st.mean(ys)
    sxx=sum((x-mx)**2 for x in xs); b=sum((x-mx)*(y-my) for x,y in zip(xs,ys))/sxx
    res=[y-(my+b*(x-mx)) for x,y in zip(xs,ys)]; se=math.sqrt(sum(r*r for r in res)/(len(p)-2)/sxx)
    return b, se
last = pts[-1][0]                       # 2026-Q1 midpoint 2026.125
win = [p for p in pts if p[0] > last - 5]   # 2021-Q2..2026-Q1 (5 calendar years = 20 quarters)
mid = last - 2.5
prev_h = [p for p in win if p[0] <= mid]; cur_h = [p for p in win if p[0] > mid]
def desc(h):
    ys=[y for _,y,_ in h]; b,se=ols(h)
    return dict(span=f'{h[0][2]}..{h[-1][2]}', n=len(h), mean=st.mean(ys), sd=st.stdev(ys), mean_time=st.mean(x for x,_,_ in h), ols=(b,se))
P, C = desc(prev_h), desc(cur_h)
print('window', win[0][2], '..', win[-1][2], 'n', len(win)); print('previous half', P); print('current half', C)
d = C['mean']-P['mean']; sep = C['mean_time']-P['mean_time']
print('mean diff %.6f  pct %.6f  sep %.6f  half-mean pace %.6f /yr  naive SE diff %.6f' % (
    d, 100*d/P['mean'], sep, d/sep, math.sqrt(P['sd']**2/P['n']+C['sd']**2/C['n'])))
print('whole-window OLS', ols(win))
print('long-run benchmark OLS (all, %s..%s, n %d)' % (pts[0][2], pts[-1][2], len(pts)), ols(pts))
print('latest', pts[-1])
