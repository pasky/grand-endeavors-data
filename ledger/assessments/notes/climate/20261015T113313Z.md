def fwd_max(A, k):
    """out[:, i] = max(A[:, i+1 : i+k+1]) for i in 0 .. L-k-1 (sparse doubling, no k-fold copy)."""
    L = A.shape[1]; m = A[:, 1:]; pw = 1
    while pw * 2 <= k:
        m = np.maximum(m[:, :-pw], m[:, pw:]); pw *= 2           # m[:, i] = max A[:, i+1 : i+1+pw]
    Lout = L - k
    return np.maximum(m[:, :Lout], m[:, k - pw:k - pw + Lout])

def b2_tcre(z):
    zz, bb = np.array(B2_pts_z), np.array(B2_pts)
    out = np.interp(z, zz, bb)
    lo = z < zz[0]; hi = z > zz[-1]
    out[lo] = bb[0] + (z[lo] - zz[0]) * (bb[1] - bb[0]) / (zz[1] - zz[0])
    out[hi] = bb[-1] + (z[hi] - zz[-1]) * (bb[-1] - bb[-2]) / (zz[-1] - zz[-2])
    return out

def describe(x):
    x = np.sort(x); n = len(x); q = lambda f: x[math.ceil(f * n) - 1]
    succ = x[np.isfinite(x)]
    out = {'p_achieved': round(float(len(succ) / n), 4)}
    for f in (0.1, 0.5, 0.9):
        v = q(f); out[f'p{int(f*100)}'] = round(float(v), 2) if np.isfinite(v) else None
    out['cond_median'] = round(float(np.median(succ)), 2) if len(succ) else None
    return out
def pct(v, qs=(10, 50, 90)):
    v = v[~np.isnan(v)]
    return [round(float(np.percentile(v, q)), 2) for q in qs] if len(v) else None
