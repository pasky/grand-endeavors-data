def kernel_H(escen=True, xscen=True, Ew=None, Xw=None, Eonly=None, Xonly=None, glo=None):
    if not escen:
        H = split_logn(stream(1, 0), *K["H"], NT)

        if Eonly is not None: w = [1.0 if i == Eonly else 0.0 for i in range(3)]

        lo = np.array([e[1] for e in K_E])[s]; hi = np.array([e[2] for e in K_E])[s]
        H = np.exp(np.log(lo) + u * np.log(hi / lo))

    return H
