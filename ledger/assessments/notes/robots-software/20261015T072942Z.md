```python
# Window statistics for the METR by-release series (shared METR fit, as of 2026-10-15).
# Reads `ledger.py series robots-software <metric> --as-of 2026-10-15`, drops month-only
# duplicate rows (YYYY-MM, which repeat a dated row), takes the running-max frontier by release
# date, and fits OLS of log2(minutes) on time (months) over the in-window points.
import subprocess, math, datetime as dt
import numpy as np
from scipy import stats

ROOT = "/home/pasky/projects/hai/grand-endeavors-agent-worktree-0005"
AS_OF = "2026-10-15"

def series(metric):
    out = subprocess.run(["uv", "run", f"{ROOT}/core/ledger.py", "series", "robots-software", metric,
                          "--as-of", AS_OF], capture_output=True, text=True, cwd=f"{ROOT}/data").stdout
    pts = []
    for line in out.strip().splitlines():
        d, v = line.split(",")[:2]
        if len(d) == 7:  # month-only duplicate
            continue
        pts.append((dt.date.fromisoformat(d), float(v)))
    pts.sort()
    fr, best = [], 0
    for d, v in pts:
        if v > best:
            fr.append((d, v)); best = v
    return fr

def yr(d):
    return d.year + (d.timetuple().tm_yday - 1) / 365.25

def ols(pts):
    x = np.array([yr(d) * 12 for d, _ in pts]); y = np.log2([v for _, v in pts])
    r = stats.linregress(x, y)
    n = len(pts); t = stats.t.ppf(0.975, n - 2)
    lo, hi = r.slope - t * r.stderr, r.slope + t * r.stderr
    dbl = 1 / r.slope
    ci = (1 / hi if hi > 0 else None, 1 / lo if lo > 0 else None)
    return dbl, ci, r

def window(fr, a, b, label):
    a, b = dt.date.fromisoformat(a), dt.date.fromisoformat(b)
    w = [(d, v) for d, v in fr if a <= d <= b]
    prior = [(d, v) for d, v in fr if d < a]
    start = prior[-1][1] if prior else w[0][1]
    last_d, last_v = w[-1]
    s_l = (yr(last_d) - yr(a)) * 12 / math.log2(last_v / start)
    dbl, ci, r = ols(w) if len(w) >= 3 else (None, (None, None), None)
    cis = f"{ci[0]:.2f}-{ci[1]:.2f}" if ci[0] and ci[1] else f"{ci[0] and round(ci[0],2)}-{ci[1] and round(ci[1],2)}"
    print(f" {label} {a}..{b} n={len(w)} start {start} -> last ({last_d}, {last_v}) x{last_v/start:.2f}; "
          f"start->last dbl {s_l:.2f} mo; OLS {dbl:.2f} CI {cis}")
    return dbl, ci, r, w

for m in ["metr-80-horizon-by-release", "metr-50-horizon-by-release"]:
    fr = series(m)
    print("==", m, len(fr), "frontier pts:", ", ".join(f"{d} {v}" for d, v in fr))
    dbl, ci, r, w = window(fr, "2024-10-15", "2026-10-15", "current")
    lastd, lastv = w[-1]
    fit = 2 ** (r.intercept + r.slope * yr(lastd) * 12)
    print(f"   OLS fitted at last pt {fit:.1f} (observed {lastv}; {math.log2(lastv/fit):+.2f} doublings)")
    window(fr, "2022-10-15", "2024-10-15", "previous")
    window(fr, "2024-10-15", "2025-10-15", "h1")
    window(fr, "2025-10-15", "2026-10-15", "h2")
    d_all, ci_all, _ = ols(fr)
    print(f" long-run all n={len(fr)} OLS {d_all:.2f} CI {ci_all[0]:.2f}-{ci_all[1]:.2f}")
```

```python
def split_logn(rng, p10, p50, p90, n):
    z = rng.standard_normal(n)
    return np.exp(np.log(p50) + np.where(z < 0, z * np.log(p50 / p10) / Z, z * np.log(p90 / p50) / Z))

def split_norm(rng, p10, p50, p90, n):
    z = rng.standard_normal(n)
    return p50 + np.where(z < 0, z * (p50 - p10) / Z, z * (p90 - p50) / Z)

def simulate(m, made=MADE, seed=20261014, fix=None, a_min=A80):
    fix = fix or {}
    rng = np.random.default_rng(seed)
    need = np.full(N, fix["need"]) if "need" in fix else split_logn(rng, *m["need"], N)
    d = np.full(N, fix["D"]) if "D" in fix else split_logn(rng, *m["D"], N)
    T = np.full(N, fix["T"]) if "T" in fix else split_norm(rng, *m["T"], N)
    lag = np.full(N, fix["lag"]) if "lag" in fix else split_logn(rng, *m["lag"], N)
    g = np.maximum(np.log2(need * 60 / a_min) + T, 0)
    eta = np.maximum(ANCHOR + g * d / 12 + lag, made)
    fail = fix.get("fail", m["fail"])
    p_ok = np.prod([1 - p for p in fail.values()])
    ok = rng.random(N) < p_ok
    eta_u = np.where(ok, eta, np.inf)
    return dict(p=p_ok, q=np.quantile(eta_u, [0.1, 0.5, 0.9]), qc=np.quantile(eta, [0.1, 0.5, 0.9]),
                by2031=np.mean(eta_u < 2032))

            f"  P(by end-2031) {r['by2031']:.2f}")
```
