def yr(d):
    n = 366 if (d.year % 4 == 0) else 365
    return d.year + (d.timetuple().tm_yday - 0.5) / n

def ols(pts):
    x = np.array([yr(d) * 12 for d, _ in pts]); y = np.log2([v for _, v in pts])
    r = stats.linregress(x, y)
    t = stats.t.ppf(0.975, len(pts) - 2)
    lo, hi = r.slope - t * r.stderr, r.slope + t * r.stderr
    return 1 / r.slope, (1 / hi if hi > 0 else None, 1 / lo if lo > 0 else None), r
