def cap_date(H, h, D):

def sim_kernel(m=K, domain="software", level=80, fixH=None, fixlag=None, fixD=None, Hdist=None):
    H = split_logn(stream(1, 0), *(Hdist or m["H"]), NT); lag = split_logn(stream(1, 1), *m["lag"], NT)
    if fixH is not None: H = np.full(NT, float(fixH))

def sim_nature(m=NAT, domain="blended", fixk=None, fixm=None, fixlag=None, fixD=None, ksplit=None):
    k = stream(2, 0).triangular(*(ksplit or m["k"]), NT)
    mm = np.abs(stream(2, 1).standard_normal(NT)) * m["m_sd"]
    lag = split_logn(stream(2, 2), *m["lag"], NT)
    if fixk is not None: k = np.full(NT, float(fixk))
    if fixm is not None: mm = np.full(NT, float(fixm))
    if fixlag is not None: lag = np.full(NT, float(fixlag))
    hW, hM, hA = nature_hW(domain)
    s0 = np.log2(hM / hW)
    # log2(H_req / h_M) = (k-1) s0 + k m ; H_req in the series' units, compared with the series anchor hA, so a constant
    # conversion offset cancels (relative mapping); the draw's own conversion error is therefore not applied here.
    G = (k - 1) * s0 + k * mm + np.log2(hM / hA)
    h, D = dom(domain)
    if fixD is not None: D = np.full(NT, float(fixD))
    eta = np.maximum(DRVT["anchor_year"] + np.maximum(G, 0) * D / 12 + lag, MADE)
    return summarize_a(eta, p_ok_of(m["fail"])), G

def sim_solo(m=SOLO, domain="blended", level=80, fixH=None, fixq=None, fixDA=None, fixDB=None, fixadopt=None, fixV=None,
             fixD=None, levelA=None, revenue_only=False, oversight_only=False, Hmult=1.0, Q=None, fixr=None):
    H = split_logn(stream(3, 0), *m["H"], NT) * Hmult
    eps = stream(3, 1).standard_normal(NT) * m["A"]["eps_sd"]
    DA = split_logn(stream(3, 2), *m["A"]["D"], NT); uA = stream(3, 3).random(NT)
    DB = split_logn(stream(3, 4), *m["B"]["D"], NT)
    adopt = split_logn(stream(3, 5), *m["adopt"], NT); V = split_logn(stream(3, 6), *m["V"], NT)
    r = stream(3, 7).uniform(*m["B"]["r"], NT) if fixr is None else np.full(NT, float(fixr))
    if fixH is not None: H = np.full(NT, float(fixH)) * Hmult
    if fixDA is not None: DA = np.full(NT, float(fixDA))
    if fixDB is not None: DB = np.full(NT, float(fixDB))
    if fixadopt is not None: adopt = np.full(NT, float(fixadopt))
    if fixV is not None: V = np.full(NT, float(fixV))
    q = m["A"]["q"] if fixq is None else fixq
    lvA = (m["A"]["level"] if levelA is None else levelA) * 2 ** eps
    TA = np.where(uA < q, dy(m["A"]["start"]) + np.maximum(np.log2(10 / lvA), 0) * DA / 12, np.inf)
    TB = dy(m["B"]["start"]) + np.log2(10 / (m["B"]["level"] * r)) * DB / 12
    Trev = np.minimum(TA, TB)
    h, D = dom(domain, level)
    if fixD is not None: D = np.full(NT, float(fixD))
    Tov = cap_date(H, h, D)
    if revenue_only: Tov = np.full(NT, -np.inf); adopt = np.zeros(NT)
    if oversight_only: Trev = np.full(NT, -np.inf)
    eta = np.maximum(np.maximum(Trev, Tov + adopt) + (m["Q"] if Q is None else Q) + V, MADE)
    return summarize_a(eta, p_ok_of(m["fail"])), Trev, Tov, adopt
