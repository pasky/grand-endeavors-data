# KPI window statistics for robots-software horizon-50-blended (as of 2026-10-16)
import subprocess, datetime as dt, math
import numpy as np
from scipy import stats

ROOT = "/home/pasky/projects/hai/grand-endeavors-agent-worktree-0005"
def series(m):
    out = subprocess.run(["uv", "run", f"{ROOT}/core/ledger.py", "series", "robots-software", m,
                          "--as-of", "2026-10-16"], capture_output=True, text=True, check=True,
                         cwd=f"{ROOT}/data").stdout
    pts = []
    for line in out.strip().splitlines():
        d, v = line.split(",")[:2]
        if len(d) != 10: continue           # month-only rows dropped
        pts.append((dt.date.fromisoformat(d), float(v)))
    return pts

def months(d, d0): return (d - d0).days / (365.25 / 12)

def ols(pts):
    d0 = pts[0][0]
    x = np.array([months(d, d0) for d, _ in pts]); y = np.log2([v for _, v in pts])
    r = stats.linregress(x, y)
    n = len(x); t = stats.t.ppf(0.975, n - 2)
    lo, hi = r.slope - t * r.stderr, r.slope + t * r.stderr
    ci = (1 / hi, (1 / lo) if lo > 0 else None)
    return 1 / r.slope, ci, r.rvalue ** 2

def endpoint(a, b):  # (date, value) pairs
    return months(b[0], a[0]) / math.log2(b[1] / a[1])

for m in ["horizon-50-blended", "horizon-50-software-eci", "horizon-50-remote-rli"]:
    pts = series(m)
    if m != "horizon-50-blended":
        pts = [p for p in pts if p[0] >= dt.date(2025, 6, 5)] if m.endswith("eci") else pts
    print(f"== {m}: n={len(pts)} {pts[0]} .. {pts[-1]}")
    db, ci, r2 = ols(pts)
    print(f"  full OLS doubling {db:.2f} mo (95% CI {ci[0]:.2f}-{ci[1]:.2f}), R2 {r2:.3f}; "
          f"endpoint {endpoint(pts[0], pts[-1]):.2f} mo; growth {pts[-1][1]/pts[0][1]:.1f}x")
    if m != "horizon-50-blended": continue
    start, last = pts[0][0], pts[-1][0]
    mid = start + (last - start) / 2
    print(f"  half split at {mid} (midpoint of measured span {start}..{last})")
    h1 = [p for p in pts if p[0] < mid]; h2 = [p for p in pts if p[0] >= mid]
    # frontier value in force at the midpoint (step function) = last h1 point
    fmid = (mid, h1[-1][1])
    for name, h, a, b in [("first", h1, pts[0], fmid), ("second", h2, fmid, pts[-1])]:
        d, c, r2 = ols(h)
        cs = f"{c[0]:.2f}-{c[1]:.2f}" if c[1] else f"{c[0]:.2f}-inf"
        print(f"  {name} half n={len(h)} {h[0][0]}..{h[-1][0]} {h[0][1]}->{h[-1][1]}: OLS {d:.2f} mo (CI {cs}, R2 {r2:.3f}); "
              f"frontier {a[1]} ({a[0]}) -> {b[1]} ({b[0]}): endpoint {endpoint(a, b):.2f} mo")
    # robustness: window extended to made_at with the unmeasured gap counted flat
    made = (dt.date(2026, 10, 16), pts[-1][1])
    print(f"  endpoint to made_at 2026-10-16 (gap counted flat, for robustness only): {endpoint(pts[0], made):.2f} mo")
    # drop latest point / largest jump
    d, c, _ = ols(pts[:-1]); print(f"  without 2026-09-22 point: OLS {d:.2f} mo")
    d, c, _ = ols([p for p in pts if p[0] != dt.date(2026, 6, 9)]); print(f"  without 2026-06-09 (RLI jump) point: OLS {d:.2f} mo")
    # split at midpoint of window to made_at
    mid2 = start + (dt.date(2026, 10, 16) - start) / 2
    g1 = [p for p in pts if p[0] < mid2]; g2 = [p for p in pts if p[0] >= mid2]
    print(f"  alt split at {mid2} (window to made_at): first n={len(g1)} OLS {ols(g1)[0]:.2f}, second n={len(g2)} OLS {ols(g2)[0]:.2f}")

# long-run benchmark context: METR's own 50% series, 2019 onward
pts = series("metr-50-horizon-by-release")
d, c, r2 = ols(pts)
print(f"== metr-50-horizon-by-release long run n={len(pts)} {pts[0][0]}..{pts[-1][0]}: OLS {d:.2f} mo (CI {c[0]:.2f}-{c[1]:.2f})")
# cross-check over the blend's window: METR's own measured 50% frontier, in force on 2025-06-05 -> last point
w = [p for p in pts if p[0] <= dt.date(2025, 6, 5)][-1]
print(f"== METR p50 cross-check: {w[1]} min in force at 2025-06-05 -> {pts[-1][1]} min ({pts[-1][0]}): "
      f"endpoint {endpoint((dt.date(2025,6,5), w[1]), pts[-1]):.2f} mo; ECI-software in force 2026-04-07 = 13.372 h vs METR 17.41 h")
# blended doubling under the remote-conversion slope sensitivity (sourced from the collector doc / drivers.yaml)

    print(f"  blend doubling with software 3.54 mo and remote {rem} mo: {2/(1/3.54+1/rem):.2f} mo")
