```python
import json,sys,calendar
AS='2026-10-28'; FROM='2025-10-28'
recs=[json.loads(l) for l in open('ledger/events/robots-software.jsonl')]
recs=[r for r in recs if r['published']<=AS]
sup=set()
for r in recs:
    s=r.get('supersedes')
    if s: sup.update(s if isinstance(s,list) else [s])
def span(d,pub):
    p=d.split('-')
    if len(p)==3: a=b=d
    elif len(p)==2:
        y,m=map(int,p); a=f'{d}-01'; b=f'{d}-{calendar.monthrange(y,m)[1]:02d}'
    else: a=f'{d}-01-01'; b=f'{d}-12-31'
    return a,min(b,pub)
mode=sys.argv[1] if len(sys.argv)>1 else 'all'
for r in sorted(recs,key=lambda r:r['date']):
    if r['id'] in sup or r.get('withdrawn') or r.get('kind')=='withdrawal': continue
    a,b=span(r['date'],r['published'])
    inw = b>=FROM and a<=AS
    if not inw: continue
    t=[x.replace('challenge:','C:').replace('milestone:','M:') for x in r['topics']]
    if mode=='all':
        print(r['id'],'|',r['date'],'|',r['kind'],r['verification']['status'],'|',','.join(t))
    elif mode in r['topics'] or mode=='full':
        print(r['id'],'|',r['date'],'pub',r['published'],'|',r['kind'],r['verification']['status'],'|',','.join(t))
        print('   ',r['claim'])
```
