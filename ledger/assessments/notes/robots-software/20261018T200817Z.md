def ols(ser, lo=None, hi=None):
    pts = [(dy(d), x) for d, x in ser if (lo is None or d >= lo) and (hi is None or d <= hi)]
    t = np.array([p[0] for p in pts]); y = np.log2([p[1] for p in pts])
    b, a = np.polyfit(t, y, 1)
    n = len(t); res = y - (a + b * t); s2 = res @ res / (n - 2); sxx = (t - t.mean()) @ (t - t.mean())
    se = np.sqrt(s2 / sxx); tq = tdist.ppf(0.975, n - 2)
    r2 = 1 - (res @ res) / ((y - y.mean()) @ (y - y.mean()))
    return dict(n=n, D=12 / b, ci=(12 / (b + tq * se), 12 / (b - tq * se)), r2=r2, a=a, b=b, s2=s2, sxx=sxx, tm=t.mean())

def draws_csv(path):

def split_logn(rng, p10, p50, p90, n):
    z = rng.standard_normal(n)
    return np.exp(np.log(p50) + np.where(z < 0, z * np.log(p50 / p10) / Z, z * np.log(p90 / p50) / Z))

def summarize_a(eta_c, p_ok):
    p = p_ok * float(np.mean(eta_c < 2101))
    q = np.array([np.quantile(eta_c, x / p_ok) if x <= p else 1e9 for x in (0.1, 0.5, 0.9)])
    return dict(p=p, p_model=float(p_ok), q=q, qc=np.quantile(eta_c, [0.1, 0.5, 0.9]),
                by2031=p_ok * float(np.mean(eta_c < 2032)), eta_c=eta_c)

def p_ok_of(fail):
    return float(np.prod([1 - p for p in fail.values()]))

    u = rng.random(n); c = np.cumsum(weights); return np.searchsorted(c / c[-1], u, side="right")

def kernel_H(escen=True, xscen=True, Ew=None, Xw=None, Eonly=None, Xonly=None, glo=None):
    if not escen:
        H = split_logn(stream(1, 0), *K["H"], NT)
    else:
        w = [e[3] for e in K_E] if Ew is None else Ew
        if Eonly is not None: w = [1.0 if i == Eonly else 0.0 for i in range(3)]
        s = scen_pick(stream(1, 2), w, NT); u = stream(1, 3).random(NT)
        lo = np.array([e[1] for e in K_E])[s]; hi = np.array([e[2] for e in K_E])[s]
        H = np.exp(np.log(lo) + u * np.log(hi / lo))
    if xscen:
        w = [x[3] for x in K_X] if Xw is None else Xw
        if Xonly is not None: w = [1.0 if i == Xonly else 0.0 for i in range(3)]
        s = scen_pick(stream(1, 4), w, NT); u = stream(1, 5).random(NT)
        his = [x[2] for x in K_X]
        if glo is not None: his[0] = -glo
        lo = np.array([x[1] for x in K_X])[s]; hi = np.array(his)[s]
        H = H * 2 ** (lo + u * (hi - lo))
    return H

def sim_kernel(escen=True, xscen=True, domain="software", level=80, fixH=None, fixlag=None, fixD=None, lag=None,
               fail=None, Hmult=1.0, **kw):
    H = kernel_H(escen, xscen, **kw) * Hmult
    if fixH is not None: H = np.full(NT, float(fixH))
    L = split_logn(stream(1, 1), *(lag or K["lag"]), NT)
    if fixlag is not None: L = np.full(NT, float(fixlag))
    h, D = dom(domain, level)
    if fixD is not None: D = np.full(NT, float(fixD))
    eta = np.maximum(cap_date(H, h, D) + L, MADE)
    return summarize_a(eta, p_ok_of(fail or K["fail"])), H

def sim_nature(lag_on=False, mdate=M_ACCEPT, hw_unc=True, spacing="scen", domain="blended", fixm=None, fixD=None,
               fixlag=None, spw=None, sponly=None, fail=None, lag=None, wdate=W_DATE, hW_fixed=None, censor=None):
    hA = {"blended": BL, "software": SW, "remote": RM}[domain][-1][1]
    if domain == "blended":
        sw_w = in_force(SW, wdate)
        lr = np.log2(remote_line(wdate)) + (stream(2, 4).standard_t(OLS_R["n"] - 2, NT) * remote_pred_sd(wdate) if hw_unc else 0.0)
        hW = np.sqrt(sw_w * 2 ** lr) * np.ones(NT)
        hM = in_force(BL, mdate)
    elif domain == "software":
        hW = np.full(NT, in_force(SW, wdate)); hM = in_force(SW, mdate)
    else:
        lr = np.log2(remote_line(wdate)) + (stream(2, 4).standard_t(OLS_R["n"] - 2, NT) * remote_pred_sd(wdate) if hw_unc else 0.0)
        hW = 2 ** lr * np.ones(NT); hM = in_force(RM, mdate)
    if hW_fixed is not None: hW = np.full(NT, float(hW_fixed))
    s0 = np.log2(hM / hW)
    ser = {"blended": BL, "software": SW, "remote": RM}[domain]
    # autonomy deficit m: half-normal sd 1 (assumed magnitude, prev's distribution and stream); CENTRAL: no lower
    # bound. Matched to the models actually tested, the shadow evaluations bound m only at or below 0 (Opus 4.8,
    # both papers: h <= 0.598 = h_M) or at most 0.73 (GPT-5.6 Sol, one paper: h <= 0.990), and only under the
    # ASSUMED transfer "failing an assigned NeurIPS question at horizon h => fully-AI main-track acceptance needs > h"
    # (review-1 MAJOR 1). Policy- / few-attempt-censored rungs give no bound (audit r2 Nature #3). Bounds = sensitivities.
    m_min = 0.0
    if censor == "frontier_jul": m_min = max(0.0, float(np.log2(in_force(ser, SHADOW) / hM)))   # draft's (rejected) reading
    elif censor == "opus48": m_min = max(0.0, float(np.log2(in_force(ser, OPUS48) / hM)))      # = 0 by construction
    elif censor == "sol56": m_min = max(0.0, float(np.log2(in_force(ser, SOL56) / hM)))        # upper limit of the bound
    elif censor == "today": m_min = max(0.0, float(np.log2(ser[-1][1] / hM)))
    elif censor == "epoch_auto": m_min = max(0.0, float(np.log2(in_force(ser, EPOCH_AUTO) / hM)))   # added 2026-10-18
    if censor is None:
        mm = np.abs(stream(2, 1).standard_normal(NT)) * NAT["m_sd"]
    else:
        from scipy.stats import norm
        a = norm.cdf(m_min / NAT["m_sd"]); mm = NAT["m_sd"] * norm.ppf(a + stream(2, 5).random(NT) * (1 - a))
    if fixm is not None: mm = np.full(NT, float(fixm))
    if spacing == "k_tri":
        k = stream(2, 0).triangular(*NAT["k"], NT); mult = k - 1
    else:
        w = [x[2] for x in N_SP] if spw is None else spw

        mult = np.array([x[1] for x in N_SP])[scen_pick(stream(2, 3), w, NT)]
    # log2(H_req / hA) = log2(hW / hA) + (1 + mult) (s0 + m)   [= prev's (k-1) s0 + k m + log2(hM/hA) when mult = k-1]

    if fixlag is not None: L = np.full(NT, float(fixlag))
    eta = np.maximum(DRVT["anchor_year"] + np.maximum(G, 0) * D / 12 + L, MADE)
    return summarize_a(eta, p_ok_of(fail or NAT["fail"])), G, hW, hM, mm

def sim_solo(adopt_on=False, gate="Lc", domain="blended", level=80, L=None, cw=None, conly=None, fixq=None, fixDB=None,
             fixV=None, fixD=None, fixr=None, Hmult=1.0, fail=None, revenue_only=False, oversight_only=False, B=None):

        c = np.array([x[1] for x in C_SC])[scen_pick(stream(3, 8), w, NT)]
        H = (L_REQ if L is None else L) * c

    eps = stream(3, 1).standard_normal(NT) * m["A"]["eps_sd"]
    DA = split_logn(stream(3, 2), *m["A"]["D"], NT); uA = stream(3, 3).random(NT)

    DB = split_logn(stream(3, 4), *Bm["D"], NT)
    adopt = split_logn(stream(3, 5), *m["adopt"], NT) if adopt_on else np.zeros(NT)
    V = split_logn(stream(3, 6), *m["V"], NT)
    r = stream(3, 7).uniform(*Bm["r"], NT) if fixr is None else np.full(NT, float(fixr))

    if oversight_only: Trev = np.full(NT, -np.inf)
    eta = np.maximum(np.maximum(Trev, Tov + adopt) + m["Q"] + V, MADE)
    return summarize_a(eta, p_ok_of(fail or m["fail"])), Trev, Tov, H
