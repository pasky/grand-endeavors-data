```python
"""rockets Export and Homestead ETAs, assess:20261015T073717Z (rubric v5); the model of
assess:20261015T060407Z (definition change d964534), unchanged, plus sensitivities only.

Starts from prev's models (assess:20261014T153125Z, eta.py there); `prev` reproduces them exactly
(same seeds, same draw order). The new-wording models add the steps the new details require and
keep every other prev input. One Monte Carlo per milestone (N=400k); every quantile and
p_achieved comes from the same distribution; sensitivities reuse the same seed (common random
numbers), changing one input at a time. Memory: a few dozen float64 vectors of 400k (<100 MB).

Year encoding (as prev): an outcome at continuous date t counts in calendar year floor(t);
quantile q = first calendar year Y with P(achieved by end of Y) >= q; null if past 2100.
This run (assess:20261015T073717Z): no new evidence; the default run reproduces assess:20261015T060407Z
exactly (same seeds, same draws) and adds sensitivities (`more`: Homestead later-programme timing,
Export trial start not before the assessment date, materials hazard start 2027).
Usage: uv run --with numpy python eta.py [prev|more]
"""
import sys
import numpy as np

N = 400_000
MADE = 2026.79  # prev's clock origin (2026-10-14); kept so the landing model reproduces prev
PREV = len(sys.argv) > 1 and sys.argv[1] == "prev"


def split_ln(rng, med, lo10, hi90, n):
    """log-split-normal with median med and 10/90% points lo10/hi90."""
    z = rng.standard_normal(n)
    sl = np.log(med / lo10) / 1.2816
    sh = np.log(hi90 / med) / 1.2816
    return med * np.exp(np.where(z < 0, z * sl, z * sh))


def quantiles(t):
    t = np.where(np.isfinite(t), t, np.inf)
    yr = np.floor(t)
    ok = yr <= 2100
    out = {"p_achieved": round(float(ok.mean()), 3)}
    srt = np.sort(yr[ok])
    for q in (0.1, 0.5, 0.9):
        k = int(np.ceil(q * len(t))) - 1
        out[f"p{int(q*100)}"] = int(srt[k]) if k < len(srt) else None
    if ok.any():
        out["cond"] = [int(np.floor(x)) for x in np.percentile(srt, [10, 50, 90])]
    return out


def raw(t):
    tt = np.where(np.isfinite(t), t, 1e9)
    return np.percentile(tt, [10, 50, 90]).round(2).tolist()


def cdf(t, years):
    yr = np.floor(np.where(np.isfinite(t), t, np.inf))
    return {y: round(float((yr <= y).mean()), 3) for y in years}


# ---------------------------------------------------------------- Export
def export(pA=0.5, start_med=1.7, start_lo=0.8, start_hi=3.6, prog_med=4.0, prog_lo=2.5, prog_hi=7.0,
           hazard=0.05, hz_start=2026.0, common=0.07, broad=0.0, start_floor=None,
           new=True, recur_p=0.6, rec_lag=(1.5, 0.5, 4.0), ph_lag=(0.5, 0.25, 1.0), seed=7):
    rng = np.random.default_rng(seed)
    start = 2025.91 + split_ln(rng, start_med, start_lo, start_hi, N)
    if start_floor is not None:  # sensitivity: no trial started before the assessment date
        start = np.maximum(start, start_floor)
    prog = split_ln(rng, prog_med, prog_lo, prog_hi, N)
    a_ok = rng.random(N) < pA
    tA = np.where(a_ok, start + prog, np.inf)
    tA = np.minimum(tA, np.where(rng.random(N) < broad, start, np.inf))  # prev sensitivity only
    if not new:  # prev: first qualifying sale at a flat hazard
        tB = hz_start + rng.exponential(1 / hazard, N)
        t = np.minimum(tA, tB)
        return np.where(rng.random(N) < common, np.inf, t)
    # new wording: recurring sales of a product line from repeated production.
    # Path B: first materials sales arrive at `hazard`; each becomes a recurring line with
    # P recur_p (else a one-off batch, which no longer counts) -> thinned Poisson at hazard*recur_p,
    # then a repeat-order lag until the sales are evidently recurring.
    tB = hz_start + rng.exponential(1 / (hazard * recur_p), N) + split_ln(rng, *rec_lag, N)
    # Path A: a marketed drug is sold continuously once launched; recurrence is evident after
    # the first repeat lot.
    tA = tA + split_ln(rng, *ph_lag, N)
    t = np.minimum(tA, tB)
    return np.where(rng.random(N) < common, np.inf, t)


# ---------------------------------------------------------------- Homestead
def homestead(gap_med=7.5, gap_lo=4.5, gap_hi=13.5, common=0.08, ab_us=0.30, ab_cn=0.25,
              d_us=(1.8, 1.3, 2.8), d_cn=(1.15, 1.0, 1.5), floor=1.0, seed=11,
              new=True, pur_us=0.35, pur_cn=0.45, clo=(10.0, 5.0, 20.0), common_new=0.11,
              later_p=0.5, later_start=(2045, 2070), later_gap=(12, 7, 20)):
    rng = np.random.default_rng(seed)
    Dus = split_ln(rng, *d_us, N)
    Dcn = split_ln(rng, *d_cn, N)
    Lus = MADE + (2028.70 - MADE) * Dus
    Lcn = MADE + (2030.00 - MADE) * Dcn
    Gus = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    Gcn = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    ab_u = rng.random(N) < ab_us
    ab_c = rng.random(N) < ab_cn
    us_base = np.where(ab_u, np.inf, Lus + Gus)   # continuously crewed base established
    cn_base = np.where(ab_c, np.inf, Lcn + Gcn)
    later_base = rng.uniform(*later_start, N) + split_ln(rng, *later_gap, N)
    u_later = rng.random(N)
    u_common = rng.random(N)
    if not new:  # prev: base with partial ISRU + 1 yr, later programme only if both abandon
        us, cn = us_base + floor, cn_base + floor
        both = ~np.isfinite(us) & ~np.isfinite(cn)
        lat = np.where(both & (u_later < later_p), later_base + floor, np.inf)
        t = np.minimum(np.minimum(us, cn), lat)
        return np.where(u_common < common, np.inf, t), np.minimum(Lus, Lcn)
    # new wording: + closure step (air, water and food from local resources and recycling, no
    # consumables resupply) + the literal self-sufficient year. Closure must be pursued.
    rng2 = np.random.default_rng(seed + 1000)  # extra draws on a separate stream: prev draws unchanged
    Cus = split_ln(rng2, *clo, N)
    Ccn = split_ln(rng2, *clo, N)
    Clat = split_ln(rng2, *clo, N)
    pu = rng2.random(N) < pur_us
    pc = rng2.random(N) < pur_cn
```
