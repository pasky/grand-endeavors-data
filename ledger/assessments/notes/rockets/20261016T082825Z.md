s = {int(l.split(",")[0]): float(l.split(",")[1]) for l in out.strip().splitlines()}
def ols(y0,y1):
    x=np.array([y for y in range(y0,y1+1)]); y=np.log([s[i] for i in x])
    A=np.vstack([x-x.mean(),np.ones_like(x)]).T
    b,res,*_=np.linalg.lstsq(A,y,rcond=None)
    r=y-A@b; se=math.sqrt((r@r)/(len(x)-2)/((x-x.mean())@(x-x.mean())))
    return b[0],se
for a,b in [(2021,2025),(2016,2020),(2011,2015),(2016,2025),(1996,2025),(1962,2025)]:
