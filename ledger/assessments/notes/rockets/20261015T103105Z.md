## milestone:the-deluge

**Wording.** "100,000 tonnes delivered to orbit in a single year"; details: "Cargo plus propellant
delivered by tanker flights counts; the launch vehicles themselves do not." Reading (prev, now codified

(`payload-mass-to-orbit`; unit "tons", metric vs short unstated).

**Gap.** 3,194.0 t (2025) vs 100,000 t: 31.31x = 4.97 doublings.

**Trend.** Launch mass is an exponential quantity (§4.1 step 3). Log-linear OLS (fit in eta-check below,
same as prev): 2021-25 slope 0.3695/yr (+44.7 %/yr, doubling 1.88 yr) -> ln(31.31)/0.3695 = 9.32 yr

def split_ln(rng, med, lo10, hi90, n):
    z = rng.standard_normal(n)
    sl = np.log(med / lo10) / 1.2816
    sh = np.log(hi90 / med) / 1.2816
    return med * np.exp(np.where(z < 0, z * sl, z * sh))


def quantiles(t):
    t = np.where(np.isfinite(t), t, np.inf)
    yr = np.floor(t)
    ok = yr <= 2100
    out = {"p_achieved": round(float(ok.mean()), 3)}
    srt = np.sort(yr[ok])
    for q in (0.1, 0.5, 0.9):
        k = int(np.ceil(q * len(t))) - 1
        out[f"p{int(q*100)}"] = int(srt[k]) if k < len(srt) else None
    if ok.any():
        out["cond"] = [int(np.floor(x)) for x in np.percentile(srt, [10, 50, 90])]
    return out


def cdf(t, years):
    yr = np.floor(np.where(np.isfinite(t), t, np.inf))
    return {y: round(float((yr <= y).mean()), 3) for y in years}


def show(name, t, years=()):

def deluge(s1=0.08, s2=0.03, s3=0.01, slope=None, stall=None, ton=1.0, seed=20261015, noise=0.07,
           prop=None):
    """Driver draw i -> simulation draw i. Returns (year, scenario code).
    ton: qualifying mass = JSR-counted mass x ton (a constant counting factor).
    prop=(s_max, y0, y1): sensitivity in which JSR's tally omits tanker-delivered propellant
    (which the framework counts) and that propellant is a share s(Y) of the qualifying mass,
    ramping linearly from 0 in year y0 to s_max in year y1 and flat after; qualifying mass in
    year Y = JSR(Y) / (1 - s(Y)), i.e. ADDED to the JSR tally (review-1 MAJOR 2)."""
    n = len(SLOPE)
    g = SLOPE if slope is None else np.full(n, slope)
    rng = np.random.default_rng(seed)
    e = rng.normal(0, noise, n)          # level noise (ln) on the crossing (own, assumed)
    u = rng.random(n)                    # failure scenario selector (own, assumed)
    u2 = rng.random(n)
    stall_len = rng.uniform(5, 15, n)
    gap = np.log(100000 / (MASS * ton))
    with np.errstate(divide="ignore", invalid="ignore"):
        t = np.where(g > 0, (gap + e) / g, np.inf)
    t = np.maximum(t, 0.0)
    if stall is not None:
        t = t + np.where(u2 < stall, stall_len, 0.0)
    yr = 2025 + np.ceil(t)

        k = np.arange(1, 76)
        s = s_max * np.clip((2025 + k - y0) / (y1 - y0), 0.0, 1.0)
        lhs = g[:, None] * k[None, :] - np.log1p(-s)[None, :]          # n x 75 (10000 x 75)
        hit = lhs >= (gap + e)[:, None]
        kk = np.where(hit.any(axis=1), k[hit.argmax(axis=1)], np.inf)
        if stall is not None:
            kk = kk + np.ceil(np.where(u2 < stall, stall_len, 0.0))
        yr = 2025 + kk
    code = np.where(u < s1, 1, np.where(u < s1 + s2, 2, np.where(u < s1 + s2 + s3, 3, 0)))
    yr = np.where(code > 0, np.inf, yr)
    code = np.where((code == 0) & (yr > 2100), 4, code)   # trend too slow to cross by 2100
    return yr, code

def export(pA=0.5, start_med=1.7, start_lo=0.8, start_hi=3.6, prog_med=4.0, prog_lo=2.5, prog_hi=7.0,
           hazard=0.05, hz_start=2026.0, common=0.07, recur_p=0.6, rec_lag=(1.5, 0.5, 4.0),
           ph_lag=(0.5, 0.25, 1.0), seed=7, codes=False):
    rng = np.random.default_rng(seed)
    start = 2025.91 + split_ln(rng, start_med, start_lo, start_hi, N)
    prog = split_ln(rng, prog_med, prog_lo, prog_hi, N)
    a_ok = rng.random(N) < pA
    tA = np.where(a_ok, start + prog, np.inf)
    tA = np.minimum(tA, np.where(rng.random(N) < 0.0, start, np.inf))  # prev's broad=0 draw, kept for draw order
    tB = hz_start + rng.exponential(1 / (hazard * recur_p), N) + split_ln(rng, *rec_lag, N)
    tA = tA + split_ln(rng, *ph_lag, N)
    t = np.minimum(tA, tB)
    uc = rng.random(N)
    t = np.where(uc < common, np.inf, t)
    if not codes:
        return t
    # 1 = C1 no product beats terrestrial alternatives (0.05/0.07 of common), 2 = C2 loss of access,
    # 3 = path A fails AND no recurring materials line by 2100, 4 = path A succeeds but only after 2100
    # and no materials line by 2100 (timing)
    c = np.zeros(N, int)
    c[uc < common * 0.05 / 0.07] = 1
    c[(uc >= common * 0.05 / 0.07) & (uc < common)] = 2
    late = (c == 0) & (np.floor(t) > 2100)
    c[late & ~a_ok] = 3
    c[late & a_ok] = 4
    return t, c

def homestead(gap_med=7.5, gap_lo=4.5, gap_hi=13.5, ab_us=0.30, ab_cn=0.25,
              d_us=(1.8, 1.3, 2.8), d_cn=(1.15, 1.0, 1.5), floor=1.0, seed=11,
              pur_us=0.35, pur_cn=0.45, clo=(10.0, 5.0, 20.0), common_new=0.11,
              later_p=0.5, later_start=(2045, 2070), later_gap=(12, 7, 20), codes=False):
    rng = np.random.default_rng(seed)
    Dus = split_ln(rng, *d_us, N)
    Dcn = split_ln(rng, *d_cn, N)
    Lus = MADE + (2028.70 - MADE) * Dus
    Lcn = MADE + (2030.00 - MADE) * Dcn
    Gus = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    Gcn = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    ab_u = rng.random(N) < ab_us
    ab_c = rng.random(N) < ab_cn
    us_base = np.where(ab_u, np.inf, Lus + Gus)
    cn_base = np.where(ab_c, np.inf, Lcn + Gcn)
    later_base = rng.uniform(*later_start, N) + split_ln(rng, *later_gap, N)
    u_later = rng.random(N)
    u_common = rng.random(N)
    rng2 = np.random.default_rng(seed + 1000)
    Cus = split_ln(rng2, *clo, N)
    Ccn = split_ln(rng2, *clo, N)
    Clat = split_ln(rng2, *clo, N)
    pu = rng2.random(N) < pur_us
    pc = rng2.random(N) < pur_cn
    us = np.where(pu, us_base + Cus + floor, np.inf)
    cn = np.where(pc, cn_base + Ccn + floor, np.inf)
    neither = ~np.isfinite(us) & ~np.isfinite(cn)
    later_on = neither & (u_later < later_p)
    lat = np.where(later_on, later_base + Clat + floor, np.inf)
    t = np.minimum(np.minimum(us, cn), lat)
    t = np.where(u_common < common_new, np.inf, t)
    if not codes:
        return t
    # failure codes (mutually exclusive, first match):
    # 1 common mode (H1-H4); 2 no lunar base ever (both programmes abandoned before a continuous
    # base) and no later programme; 3 lunar base(s) built but neither pursues food closure, and no
    # later programme; 4 a later programme forms but closes only after 2100, or a pursuing lunar
    # programme closes only after 2100 (timing)
    c = np.zeros(N, int)
    fail = np.floor(np.where(np.isfinite(t), t, np.inf)) > 2100
    c[fail & (u_common < common_new)] = 1
    rest = fail & (c == 0)
    no_base = ab_u & ab_c
    pursued_lunar = (pu & ~ab_u) | (pc & ~ab_c)
    c[rest & ~later_on & no_base] = 2
    c[rest & ~later_on & ~no_base & ~pursued_lunar] = 3
    c[rest & (c == 0)] = 4
    return t, c
