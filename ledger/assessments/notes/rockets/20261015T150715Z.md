"""rockets milestone:the-homestead ETA, assess:20261015T150715Z (rubric v6, audit_correction
2026-10-15-anchors). homestead() is prev's (assess:20261015T103105Z eta.py, itself
assess:20261015T073717Z's) verbatim: same seeds, same draw order, N = 400,000.
Year encoding: outcome at continuous date t counts in calendar year floor(t); quantile q = first
calendar year Y with P(achieved by end of Y) >= q; null past 2100.
Usage: uv run --with numpy python eta.py   (ulimit -v 4000000; ~20 float64 vectors of 400k)
"""
import sys

import numpy as np

N = 400_000
MADE = 2026.79  # prev's clock origin (2026-10-15), kept so the landing model reproduces prev

def split_ln(rng, med, lo10, hi90, n):
    z = rng.standard_normal(n)
    sl = np.log(med / lo10) / 1.2816
    sh = np.log(hi90 / med) / 1.2816
    return med * np.exp(np.where(z < 0, z * sl, z * sh))


def quantiles(t):
    t = np.where(np.isfinite(t), t, np.inf)
    yr = np.floor(t)
    ok = yr <= 2100
    out = {"p_achieved": round(float(ok.mean()), 3)}
    srt = np.sort(yr[ok])
    for q in (0.1, 0.5, 0.9):
        k = int(np.ceil(q * len(t))) - 1
        out[f"p{int(q*100)}"] = int(srt[k]) if k < len(srt) else None
    if ok.any():
        out["cond"] = [int(np.floor(x)) for x in np.percentile(srt, [10, 50, 90])]
    return out


def cdf(t, years):
    yr = np.floor(np.where(np.isfinite(t), t, np.inf))
    return {y: round(float((yr <= y).mean()), 3) for y in years}


def show(name, t, years=()):
    print(f"{name}: {quantiles(t)}", (cdf(t, years) if years else ""))


# ---------------------------------------------------------------- Homestead (prev, verbatim + codes)
def homestead(gap_med=7.5, gap_lo=4.5, gap_hi=13.5, ab_us=0.30, ab_cn=0.25,
              d_us=(1.8, 1.3, 2.8), d_cn=(1.15, 1.0, 1.5), floor=1.0, seed=11,
              pur_us=0.35, pur_cn=0.45, clo=(10.0, 5.0, 20.0), common_new=0.11,
              later_p=0.5, later_start=(2045, 2070), later_gap=(12, 7, 20), codes=False):
    rng = np.random.default_rng(seed)
    Dus = split_ln(rng, *d_us, N)
    Dcn = split_ln(rng, *d_cn, N)
    Lus = MADE + (2028.70 - MADE) * Dus
    Lcn = MADE + (2030.00 - MADE) * Dcn
    Gus = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    Gcn = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    ab_u = rng.random(N) < ab_us
    ab_c = rng.random(N) < ab_cn
    us_base = np.where(ab_u, np.inf, Lus + Gus)
    cn_base = np.where(ab_c, np.inf, Lcn + Gcn)
    later_base = rng.uniform(*later_start, N) + split_ln(rng, *later_gap, N)
    u_later = rng.random(N)
    u_common = rng.random(N)
    rng2 = np.random.default_rng(seed + 1000)
    Cus = split_ln(rng2, *clo, N)
    Ccn = split_ln(rng2, *clo, N)
    Clat = split_ln(rng2, *clo, N)
    pu = rng2.random(N) < pur_us
    pc = rng2.random(N) < pur_cn
    us = np.where(pu, us_base + Cus + floor, np.inf)
    cn = np.where(pc, cn_base + Ccn + floor, np.inf)
    neither = ~np.isfinite(us) & ~np.isfinite(cn)
    later_on = neither & (u_later < later_p)
    lat = np.where(later_on, later_base + Clat + floor, np.inf)
    t = np.minimum(np.minimum(us, cn), lat)
    t = np.where(u_common < common_new, np.inf, t)
    if not codes:
        return t
    # failure codes (mutually exclusive, first match):
    # 1 common mode (H1-H4); 2 no lunar base ever (both programmes abandoned before a continuous
    # base) and no later programme; 3 lunar base(s) built but neither pursues food closure, and no
    # later programme; 4 a later programme forms but closes only after 2100, or a pursuing lunar
    # programme closes only after 2100 (timing)
    c = np.zeros(N, int)
    fail = np.floor(np.where(np.isfinite(t), t, np.inf)) > 2100
    c[fail & (u_common < common_new)] = 1
    rest = fail & (c == 0)
    no_base = ab_u & ab_c
    pursued_lunar = (pu & ~ab_u) | (pc & ~ab_c)
    c[rest & ~later_on & no_base] = 2
    c[rest & ~later_on & ~no_base & ~pursued_lunar] = 3
    c[rest & (c == 0)] = 4
    return t, c


def decompose(name, t, c, labels):
    yr = np.floor(np.where(np.isfinite(t), t, np.inf))
    print(f"{name}: P(fail by 2100) {float((yr > 2100).mean()):.3f} =",
          ", ".join(f"{labels[k]} {float((c == k).mean()):.3f}" for k in sorted(labels)))



if __name__ == "__main__":
    print("=== HOMESTEAD (prev model and inputs, kept: pursuit US 0.35 / China 0.45)")
    t, c = homestead(codes=True)
    show("base", t, (2030, 2035, 2040, 2045, 2049, 2050, 2060, 2070, 2075, 2080, 2100))
    decompose("failure", t, c, {1: "common mode H1-H4", 2: "no base ever + no later programme",
                                3: "bases on resupply (no pursuit) + no later programme",
                                4: "closure only after 2100"})
    show("pursue US 1/7 (station outcome rate transferred)/0.45", homestead(pur_us=1/7))
    show("pursue 1/7 / 1/7 (station outcome rate transferred to both)", homestead(pur_us=1/7, pur_cn=1/7))
    show("pursue 0.2/0.3", homestead(pur_us=0.2, pur_cn=0.3))
    show("pursue 0.5/0.6", homestead(pur_us=0.5, pur_cn=0.6)); show("pursue 0.7/0.8", homestead(pur_us=0.7, pur_cn=0.8))
    show("later programme P 0.25 (class 1/4)", homestead(later_p=0.25))
    show("later programme P 0.3", homestead(later_p=0.3)); show("later programme P 0.7", homestead(later_p=0.7))
    show("abandon 0.50/0.40", homestead(ab_us=0.5, ab_cn=0.4))
    show("abandon 0.67/0.67 (high side: 0 of 3 ended crewed lunar programmes built a base)", homestead(ab_us=2/3, ab_cn=2/3))
    show("common 0.06", homestead(common_new=0.06)); show("common 0.20", homestead(common_new=0.20))
    show("closure 5 yr (80% 2.5-10)", homestead(clo=(5.0, 2.5, 10.0)))
    show("closure 15 yr (80% 8-30)", homestead(clo=(15.0, 8.0, 30.0)))
    show("landing->base 5 yr", homestead(gap_med=5, gap_lo=3, gap_hi=9))
    show("landing->base 12 yr", homestead(gap_med=12, gap_lo=7, gap_hi=20))
    show("US discount 2.5", homestead(d_us=(2.5, 1.8, 3.5)))
    show("later start U(2035,2055)", homestead(later_start=(2035, 2055)))
    show("later start U(2055,2085)", homestead(later_start=(2055, 2085)))
    show("later to base LN(8; 5-13)", homestead(later_gap=(8, 5, 13)))
    show("later to base LN(18; 10-30)", homestead(later_gap=(18, 10, 30)))

    # ---- written and run AFTER the pre-anchor line was in eta-notes.md (RUBRIC.md §4.1 step 4)
