def split_ln(rng, med, lo10, hi90, n, lower=None):
    """prev's split lognormal; `lower` (optional) truncates it from below at `lower` (same normal draws,
    mapped through the inverse CDF, so the random stream and draw order are unchanged)."""
    z = rng.standard_normal(n)
    sl = np.log(med / lo10) / 1.2816
    sh = np.log(hi90 / med) / 1.2816
    if lower is not None:
        zb = np.log(lower / med) / (sl if lower < med else sh)
        fb = ndtr(zb)
        z = ndtri(fb + ndtr(z) * (1 - fb))
    return med * np.exp(np.where(z < 0, z * sl, z * sh))


def quantiles(t):

    return out

def export(pA=0.5, start_med=1.7, start_lo=0.8, start_hi=3.6, prog_med=4.0, prog_lo=2.5, prog_hi=7.0,
           hazard=0.05, hz_start=2026.0, common=0.07, recur_p=0.6, rec_lag=(1.5, 0.5, 4.0),
           ph_lag=(0.5, 0.25, 1.0), seed=7, codes=False, start_min=None):
    """prev's export() verbatim, plus start_min: path A's trial start conditioned on >= start_min
    (the split lognormal of the delay from 2025.91 truncated at start_min - 2025.91)."""
    rng = np.random.default_rng(seed)
    start = 2025.91 + split_ln(rng, start_med, start_lo, start_hi, N,
                               lower=None if start_min is None else start_min - 2025.91)
    prog = split_ln(rng, prog_med, prog_lo, prog_hi, N)
    a_ok = rng.random(N) < pA
    tA = np.where(a_ok, start + prog, np.inf)
    tA = np.minimum(tA, np.where(rng.random(N) < 0.0, start, np.inf))  # prev's broad=0 draw, kept for draw order
    tB = hz_start + rng.exponential(1 / (hazard * recur_p), N) + split_ln(rng, *rec_lag, N)
    tA = tA + split_ln(rng, *ph_lag, N)
    t = np.minimum(tA, tB)
    uc = rng.random(N)
    t = np.where(uc < common, np.inf, t)
    if not codes:
        return t
    c = np.zeros(N, int)
    c[uc < common * 0.05 / 0.07] = 1
    c[(uc >= common * 0.05 / 0.07) & (uc < common)] = 2
    late = (c == 0) & (np.floor(t) > 2100)
    c[late & ~a_ok] = 3
    c[late & a_ok] = 4
    return t, c, tA, tB

def homestead(gap_med=7.5, gap_lo=4.5, gap_hi=13.5, ab_us=0.30, ab_cn=0.25,
              d_us=(1.8, 1.3, 2.8), d_cn=(1.15, 1.0, 1.5), floor=1.0, seed=11,
              pur_us=0.35, pur_cn=0.45, clo=(10.0, 5.0, 20.0), common_new=0.11,
              later_p=0.5, later_start=(2045, 2070), later_gap=(12, 7, 20), codes=False,
              made=MADE_PREV, us_claim=2028.70, cn_claim=2030.00):
    """prev's homestead() verbatim; the clock (made) and the landing claims are now parameters."""
    rng = np.random.default_rng(seed)
    Dus = split_ln(rng, *d_us, N)
    Dcn = split_ln(rng, *d_cn, N)
    Lus = made + (us_claim - made) * Dus
    Lcn = made + (cn_claim - made) * Dcn
    Gus = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    Gcn = split_ln(rng, gap_med, gap_lo, gap_hi, N)
    ab_u = rng.random(N) < ab_us
    ab_c = rng.random(N) < ab_cn
    us_base = np.where(ab_u, np.inf, Lus + Gus)
    cn_base = np.where(ab_c, np.inf, Lcn + Gcn)
    later_base = rng.uniform(*later_start, N) + split_ln(rng, *later_gap, N)
    u_later = rng.random(N)
    u_common = rng.random(N)
    rng2 = np.random.default_rng(seed + 1000)
    Cus = split_ln(rng2, *clo, N)
    Ccn = split_ln(rng2, *clo, N)
    Clat = split_ln(rng2, *clo, N)
    pu = rng2.random(N) < pur_us
    pc = rng2.random(N) < pur_cn
    us = np.where(pu, us_base + Cus + floor, np.inf)
    cn = np.where(pc, cn_base + Ccn + floor, np.inf)
    neither = ~np.isfinite(us) & ~np.isfinite(cn)
    later_on = neither & (u_later < later_p)
    lat = np.where(later_on, later_base + Clat + floor, np.inf)
    t = np.minimum(np.minimum(us, cn), lat)
    t = np.where(u_common < common_new, np.inf, t)
    if not codes:
        return t
    c = np.zeros(N, int)
    fail = np.floor(np.where(np.isfinite(t), t, np.inf)) > 2100
    c[fail & (u_common < common_new)] = 1
    rest = fail & (c == 0)
    no_base = ab_u & ab_c
    pursued_lunar = (pu & ~ab_u) | (pc & ~ab_c)
    c[rest & ~later_on & no_base] = 2
    c[rest & ~later_on & ~no_base & ~pursued_lunar] = 3
    c[rest & (c == 0)] = 4
    return t, c, Lus, Lcn
