"""the-long-middle-age ETA, assess run 20261015T145816Z (rubric v6, audit_correction 2026-10-15-anchors), as of
2026-10-15. Mid class verbatim from prev (run 20261015T123947Z, notes Appendix A), same seed 404 and draw order,
so prev reproduces exactly. Quantiles: sorted times, index ceil(q*N)-1, failures and >2100 kept as +inf
(unconditional); quantile year = calendar year (floor). Run: uv run python eta.py (stdlib; ulimit -v 1048576)."""
import math, random, sys
N = 40000
MADE = 2026 + (31+28+31+30+31+30+31+31+30+14)/365  # 2026-10-15
def q_of(ts, q):
    s = sorted(ts); v = s[math.ceil(q*len(s))-1]
    return None if v > 2100 else round(v, 2)
def summ(ts):
    p = sum(1 for t in ts if t <= 2100)/len(ts)
    succ = sorted(t for t in ts if t <= 2100)
    cm = succ[len(succ)//2] if succ else None
    return dict(p=round(p, 4), p10=q_of(ts, .1), p50=q_of(ts, .5), p90=q_of(ts, .9),
                cond_med=None if cm is None else round(cm, 2),
                P_by_2039end=round(sum(1 for t in ts if t < 2040)/len(ts), 4))
def ln(med, sd, z): return med*math.exp(sd*z)

class Mid:   # verbatim from prev
    def __init__(s):
        g=random.Random(404)
        s.d=[dict(za=g.gauss(0,1), zb=g.gauss(0,1), u1=g.random(), u2=g.random()) for _ in range(N)]
    def run(s, Amed=8, Asd=.65, Amin=3, Bmed=6, Bsd=.30, Bmin=4, f1=.15, f2=.12, generic=None, anchor=2025):
        ts=[]
        for d in s.d:
            if generic is not None:
                if d['u1']<generic: ts.append(math.inf); continue
            else:
                if d['u1']<f1 or d['u2']<f2: ts.append(math.inf); continue
            A=max(Amin,ln(Amed,Asd,d['za'])); B=max(Bmin,ln(Bmed,Bsd,d['zb']))
            ts.append(max(anchor+A+B,MADE))
        return summ(ts)
    def stageA(s, Amed=8, Asd=.65, Amin=3, f2=.12, anchor=2025, by=2040):
        # P(a qualifying >=10-yr clock+function intervention exists by 'by', no incidence validation needed)
        k=0
        for d in s.d:
            if d['u2']<f2: continue
            if anchor+max(Amin,ln(Amed,Asd,d['za'])) < by: k+=1
        return round(k/N,4)

def show(tag, res): print(f"  {tag:58s} {res}")
def dep_sens(D):
    # F1/F2 dependence (prev review minor): comonotone (u2:=u1, failure = max) and countermonotone (u2:=1-u1, failure = sum)
    import copy
    for tag, f in [('F1/F2 comonotone (failure max(.15,.12)=.15)', lambda d: d['u1']),
                   ('F1/F2 countermonotone (failure .15+.12=.27)', lambda d: 1-d['u1'])]:
        E = copy.copy(D); E.d = [dict(d, u2=f(d)) for d in D.d]
        show(tag, E.run())

def anchor_compare(D):
    print('== anchor comparison (manifold-60n0viotdzfv1f9jmm1q-p, 0.6012 by 2039-12-31)')
    show('ours, full milestone, P(by end-2039)', D.run()['P_by_2039end'])

    show('stage A, A median 5', D.stageA(Amed=5))
    show('stage A, A median 12', D.stageA(Amed=12))

        show(f'  full milestone, A median {am}', D.run(Amed=am))

if __name__ == "__main__":
    D = Mid()
    show('prev reproduction = central (F1 .15, F2 .12)', D.run())
    for tag, kw in [('F1 .111', dict(f1=1/9)), ('F1 .182', dict(f1=.1818)), ('F1 .05', dict(f1=.05)),
                    ('F1 .30 (proportional reading)', dict(f1=.30)), ('F1 .60 (human class binding)', dict(f1=.60)),
                    ('F2 .111', dict(f2=1/9)), ('F2 .05', dict(f2=.05)), ('F2 .25', dict(f2=.25)),
