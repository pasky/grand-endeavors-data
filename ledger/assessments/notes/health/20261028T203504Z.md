## Scripts

Window listing (`uv run`-free, plain python3, run from the data repo root; argument = challenge slug, or
`untagged`):

```python
import json,sys
recs=[json.loads(l) for l in open('ledger/events/health.jsonl')]
recs=[r for r in recs if r['published']<='2026-10-28']
sup=set()
for r in recs:
    s=r.get('supersedes')
    if s: sup.update(s if isinstance(s,list) else [s])
eff=[r for r in recs if r['id'] not in sup and not r.get('withdrawn')]
def inwin(d,pub):
    if len(d)==4: end=d+'-12-31'; start=d+'-01-01'
    elif len(d)==7: start=d+'-01'; end=d+'-31'
    else: start=end=d[:10]
    end=min(end,pub)
    return end>='2025-10-28' and start<='2026-10-28'
tag=sys.argv[1] if len(sys.argv)>1 else None
n=0
for r in sorted(eff,key=lambda r:r['date']):
    if not inwin(r['date'],r['published']): continue
    if tag=='untagged':
        if any(t.startswith('challenge:') for t in r['topics']): continue
    elif tag and ('challenge:'+tag) not in r['topics']: continue
    n+=1
    print(f"{r['id']} | {r['date']} pub {r['published']} | {r['kind']} | {r['verification']['status']} | {','.join(r['topics'])}\n   {r['claim']}")
print(n,'events',file=sys.stderr)
```
