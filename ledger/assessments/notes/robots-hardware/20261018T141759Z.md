import sys, csv, io, subprocess, datetime as dt
from math import comb
R = "/home/pasky/projects/hai/grand-endeavors-agent-worktree-0002"
sys.path.insert(0, f"{R}/core")
import ledger
AS_OF = "2026-10-18"
# 1) the registered series (month-end counts at work + today's day point)
out = subprocess.run(["uv", "run", f"{R}/core/ledger.py", "series", "robots-hardware", "gp-deployments",
                      "--as-of", AS_OF], capture_output=True, text=True, check=True).stdout
s = {r[0]: int(float(r[1])) for r in csv.reader(io.StringIO(out)) if r}
print("series from 2024-10:", {k: v for k, v in s.items() if k >= "2024-10"})
# 2) exact day roster from the effective events (the same function the collector uses)
evs = [e for e in ledger.effective_events(ledger.events("robots-hardware"), dt.date.fromisoformat(AS_OF))
       if str(e["published"]) <= AS_OF]
cases, tl = ledger.deployment_roster(evs, AS_OF)
print("\ncases:")
for c in cases:
    print(f"  {c['maker']} / {c['customer']}: since {c['since']} ({c['first_event']}), status {c['status']}, "
          f"counted {c['counted']}, updates {[(u['date'], u['status'], u['event']) for u in c['updates']]}")

    print(f"  {t['date']}: count {t['count']} +{t['added']} -{t['removed']}")
def count_on(d):
    c = 0
    for t in tl:
        if str(t["date"]) <= d: c = t["count"]
    return c
def flows(a, b):
    add = sum(len(t["added"]) for t in tl if a < str(t["date"]) <= b)
    rem = sum(len(t["removed"]) for t in tl if a < str(t["date"]) <= b)
    return add, rem

print(f"\ncount on {A}: {count_on(A)}, {B}: {count_on(B)}, {C}: {count_on(C)}")

      "added/removed:", [flows(x, y) for x, y in ((A, M1), (M1, B), (B, M2), (M2, C))])
print(f"month-end check: 2025-10 {s['2025-10']} -> day point {C} {s[C]} = net {s[C]-s['2025-10']}; "
      f"2024-10 {s['2024-10']} -> 2025-10 {s['2025-10']} = net {s['2025-10']-s['2024-10']}")
# 3) noise: entries are Poisson; test equal entry rates (conditional binomial) and the ratio's CI
def cdf(k, N, q): return sum(comb(N, i) * q**i * (1-q)**(N-i) for i in range(k + 1))
def bisect(f, inc=True):
    a, b = 0.0, 1.0
    for _ in range(60):
        m = (a + b) / 2
        if (f(m) > 0) == inc: b = m
        else: a = m
    return (a + b) / 2
def stats(c, p):
    N = c + p
    pv = sum(comb(N, k) for k in range(c, N + 1)) / 2 ** N
    res = {"cur": c, "prev": p, "ratio": round(c/p, 2) if p else None, "p_one_sided": round(pv, 3)}
    for a in (0.2, 0.05):
        lo = 0 if c == 0 else bisect(lambda q: 1 - cdf(c - 1, N, q) - a / 2)
        hi = 1 if c == N else bisect(lambda q: cdf(c, N, q) - a / 2, inc=False)
        res[f"ci{int((1-a)*100)}"] = (round(lo/(1-lo), 2), round(hi/(1-hi), 2) if hi < 1 else None)
    return res
(ca, cr), (pa, pr) = flows(B, C), flows(A, B)
print("entries (Poisson) cur vs prev:", stats(ca, pa))
# P(net_cur > net_prev) is not testable exactly with one exit; Monte Carlo under equal entry rates:
import random
random.seed(1)
lam = (ca + pa) / 2; mu_exit = (cr + pr) / 2
def pois(l):
    k, t = 0, random.expovariate(1)
    while t < l: k += 1; t += random.expovariate(1)
    return k
n = 200000; ge = 0
for _ in range(n):
    if pois(lam) - pois(mu_exit) - (pois(lam) - pois(mu_exit)) >= cur_net - prev_net: ge += 1
print(f"MC P(net difference >= {cur_net - prev_net} | equal entry rate {lam}, exit rate {mu_exit}/yr) = {ge/n:.3f}")
# 4) sensitivity: borderline counted deployments under the new wording
border = {"Spirit AI / CATL": "maker is a CATL investee", "Leju Robotics / Zhaofeng Mechanical & Electronic":
          "JV partner; billed as a training demonstration", "AgiBot / SAIC Motor": "maker is an SAIC Financial investee"}
def since(lbl):
    return next(str(c["since"]) for c in cases if f"{c['maker']} / {c['customer']}" == lbl)
for lbl, why in border.items():
    d = since(lbl); dc = 1 if B < d <= C else 0; dp = 1 if A < d <= B else 0
    print(f"drop {lbl} ({why}): net {cur_net - dc} vs {prev_net - dp}")
dc = sum(1 for l in border if B < since(l) <= C)
print(f"drop all three: net {cur_net - dc} vs {prev_net}")
# Figure/BMW entered in the previous window and left in the current one: never counting it
# removes +1 from the previous window and the -1 from the current one
print("Figure/BMW never counted (2025 job read as a non-production test, as BMW said in 2026):",
      f"net {cur_net + 1} vs {prev_net - 1}; with all three above also dropped: net {cur_net + 1 - dc} vs {prev_net - 1}")
print("Humanoid/Siemens: working and ended reports the same day (2026-01-15), so it never enters the count;",
      f"had the end been reported later it would add +1 and -1 inside the current window: net still {cur_net}")

### Verdict

Green (On track): 9 deployments at work on 2026-10-18; net +6 in the last 12 months vs +3 the 12
months before. This is at or above the tier-3 benchmark, so the status is unchanged from prev.

Caveats (reported, not scored):
- Coverage comes from one retrospective sweep. Older cases, most likely Chinese ones, may be missing,
  and a backfill of 2024–25 deployments would cut the margin.
- Autonomy is as reported by companies, with no figures on interventions.
- Ends are reported less often than starts. Unreported ends would inflate the current count, and the
  2026 additions have had the least time to be reported ended.

What would change it:
- next year adding net fewer than +6 → yellow if the count is still rising;
- reports ending or disputing two or more 2026 deployments → the margin vanishes;
- a strict "affiliated maker" ruling by the owner → 3 vs 3.
