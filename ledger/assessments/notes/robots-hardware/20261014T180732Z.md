# robots-hardware milestone ETA Monte Carlo, as of 2026-10-14 (rubric v5; r1 model of assess:20261014T145110Z + audit 2026-10-14-r2 fixes
# + audit 2026-10-14-r3 finding 2: the shared-stall sensitivity now changes the shared threshold S0 itself)

import numpy as np, datetime as dt, sys

MA = 2026 + (dt.date(2026, 10, 14) - dt.date(2026, 1, 1)).days / 365.0   # 2026.784
Z80 = 1.2816
M_MISS = 0.15   # P(a qualifying report published before made_at was missed by gather) - assumed


def lnorm(rng, med, lo, hi, n):
    s = (np.log(hi) - np.log(lo)) / (2 * Z80)
    return med * np.exp(s * rng.standard_normal(n))


def odds_lnorm(rng, pmed, plo, phi, n):
    o = lambda p: p / (1 - p)
    return lnorm(rng, o(pmed), o(plo), o(phi), n)


def summarize(eta, fail, w=None):
    """eta: years (inf = never); fail: bool mask of failure-scenario draws; w: weights.
    Samples with eta < made_at mean a qualifying report should already exist: none is in the
    ledger, so they keep weight M_MISS (missed by gather) and are dated made_at."""
    eta = eta.copy()
    w = np.ones_like(eta) if w is None else w.copy()
    early = eta < MA
    w[early] *= M_MISS
    eta[early] = MA
    eta[fail] = np.inf
    eta[eta > 2100] = np.inf
    o = np.argsort(eta)
    e, cw = eta[o], np.cumsum(w[o]) / w.sum()
    def q(p):
        i = np.searchsorted(cw, p)
        return None if i >= len(e) or not np.isfinite(e[i]) else round(float(e[i]), 2)
    fin = np.isfinite(eta)
    pa = float(w[fin].sum() / w.sum())
    ok = ~fail & np.isfinite(eta)
    oc = np.argsort(eta[ok]); ec, cwc = eta[ok][oc], np.cumsum(w[ok][oc]) / w[ok].sum()
    cond = [round(float(ec[np.searchsorted(cwc, p)]), 2) for p in (0.1, 0.5, 0.9)]
    return dict(p_achieved=round(pa, 3), p10=q(0.1), p50=q(0.5), p90=q(0.9), cond=cond,
                early_share=round(float(early.mean()), 3),
                P5=round(float(w[eta <= MA + 5].sum() / w.sum()), 3),
                P10=round(float(w[eta <= MA + 10].sum() / w.sum()), 3))


def grid_first(hfun, jfun, n, t0, t1=2100.0, dtg=0.05, chunk=1500, seed=1):
    """First-success time for a per-sample hazard on a time grid.
    hfun(idx, t) -> continuous hazard (1/yr), shape (len(idx), len(t));
    jfun(idx, t) -> discrete -log(1-q) at grid points (or None)."""
    rng = np.random.default_rng(seed)
    t = np.arange(t0, t1 + dtg, dtg)
    out = np.full(n, np.inf)
    for a in range(0, n, chunk):
        idx = np.arange(a, min(n, a + chunk))
        H = hfun(idx, t) * dtg
        if jfun is not None:
            H = H + jfun(idx, t)
        H = np.cumsum(H, axis=1)
        E = rng.exponential(size=len(idx))[:, None]
        hit = H >= E
        anyhit = hit.any(axis=1)
        first = np.argmax(hit, axis=1)
        out[idx[anyhit]] = t[first[anyhit]]
    return out


# ---------------- shared draws (correlated across the four humanoid milestones) ----------------
def shared(rng, n, T_med=7.62, T_lo=4.0, T_hi=14.0, s0=0.02):
    T = lnorm(rng, T_med, T_lo, T_hi, n) / 12.0          # failure-fraction halving, years (Dyna, mapping assumed)
    T_odds = T * (6.52 / 7.62)                             # failure-odds halving (Dyna: odds 1/3 -> 0.075)
    S0 = rng.random(n) < s0                                # shared stall scenario (assumed 0.02); r3: threshold is a
                                                           # parameter, same uniform draw, so a sensitivity moves only it
    MTBF = lnorm(rng, 16, 5, 60, n)                        # h, best real-line humanoid (AgiBot G2 'two-shift')
    I_task = lnorm(rng, 0.15, 0.08, 0.5, n)                # task failures / 8h robot-shift (Longcheer arithmetic)
    f_h = np.minimum(lnorm(rng, 0.5, 0.2, 1.0, n), 1.0)    # share of task failures needing a human (assumed)
    I_oth = lnorm(rng, 0.3, 0.05, 1.5, n)                  # other human assists / shift (assumed)
    return dict(T=T, T_odds=T_odds, S0=S0, I_hw=8.0 / MTBF, I_task=I_task, f_h=f_h, I_oth=I_oth)


# ---------------- Robotaxi ----------------
def robotaxi(n=400_000, seed=20261014, reading="A", g_med=0.88, g_sd=0.30, R_med=1000, lag=(0.4, 0.15, 1.2),
             p_fail=0.05, H=None, g_floor=0.15, onroad=None):
    # g_med 0.88 = ln(450/250)/(8/12): MEASURED point (two legacy Waymo endpoints). g_sd 0.30 and g_floor 0.15
    # are the ASSUMED forecasting spread (two endpoints measure no spread) - audit 2026-10-14-r2 finding 5.
    rng = np.random.default_rng(seed)
    T0 = 2025.92
    R = lnorm(rng, R_med, R_med * 0.7, R_med * 1.3, n)
    u = rng.uniform(0.7, 0.9, n)                             # on-road share (assumed)
    R = R * (u if onroad is None else onroad)                # SF robotaxis on road
    if reading == "A":
        Hh = lnorm(rng, *(H or (5000, 3500, 8000)), n)       # taxis + ride-hail on road at peak
        r0 = R / Hh
    elif reading == "B":
        r0 = lnorm(rng, 0.9, 0.6, 1.4, n)                    # licensed taxis only
    else:
        r0 = lnorm(rng, 0.04, 0.025, 0.07, n)                # all active ride-hail vehicles in a month
    g = np.maximum(g_med + g_sd * rng.standard_normal(n), g_floor)
    L = lnorm(rng, *lag, n) if lag[0] > 0 else np.zeros(n)
    eta = T0 + np.maximum(0, np.log(1 / r0)) / g + L
    fail = rng.random(n) < p_fail
    return summarize(eta, fail)

    rA = lnorm(rng, 1000, 700, 1300, n) * 0.8 / lnorm(rng, 5000, 3500, 8000, n)

def dark_line(n=400_000, seed=20261014, sh=None, n_act=7, K=3, a=(0.6, 0.3, 0.85), pco=(0.3, 0.1, 0.6),
              k=6.0, setup=(0.4, 0.15, 1.2), lag=(0.15, 0.03, 0.6), cond_mult=1.0, task_mult=1.0, p_fail_own=0.10,
              T_scale=1.0, debug=False,
              throughput=True, b0=(0.7, 0.4, 1.0), avail=(0.95, 0.88, 0.99), T_tp_ratio=8.10 / 7.62, tp_scale=1.0,
              charging=True, cfrac=(0.10, 0.02, 0.25), sh_mod=None):
    """r2 additions (audit 2026-10-14-r2 finding 3), drawn from a SEPARATE generator so the r1 draws are unchanged:
    - throughput gate: the week counts only once the bottleneck station's throughput ratio b(t) x station
      availability >= 0.90 of the line's required throughput; b(t) = b0 x 2^((t-2026.5)/T_tp), T_tp = shared
      reliability halving x 8.10/7.62 (Dyna per-shift output 480 -> 1,590 in 14 mo = 8.10-mo doubling; same
      event, so fully correlated with the reliability draw; mapping onto factory stations assumed).
    - routine self-maintenance (charging/battery swap) done by robots: robots charging at any moment occupy
      round(n_act x cfrac) of the K spares, so K_eff = K - that (spares left to absorb faults)."""
    rng = np.random.default_rng(seed)
    rng2 = np.random.default_rng(seed + 11)
    sh = sh or shared(np.random.default_rng(seed + 1), n)
    if sh_mod is not None:
        sh = sh_mod(dict(sh), np.random.default_rng(seed + 13), n)
    Hh = n_act * 168.0
    B0 = lnorm(rng2, *b0, n)
    AV = np.minimum(lnorm(rng2, *avail, n), 0.999)
    CF = lnorm(rng2, *cfrac, n)
    K_eff = np.maximum(K - np.round(n_act * CF), 0).astype(int) if charging else np.full(n, K)
    T_tp = sh["T"] * T_tp_ratio * T_scale * tp_scale
    t_tp = 2026.5 + np.maximum(0, np.log2(0.90 / (B0 * AV))) * T_tp if throughput else np.full(n, -np.inf)
    A = np.minimum(lnorm(rng, *a, n), 0.95) if a[0] > 0 else np.zeros(n)
    r_hw = sh["I_hw"] / 8 * cond_mult
    r_t = (sh["f_h"] * sh["I_task"] + sh["I_oth"]) / 8 * cond_mult * task_mult
    lam_na0 = Hh * (r_hw * (1 - A) + r_t)
    lam_m0 = Hh * r_hw * A
    oco0 = odds_lnorm(rng, *pco, n)
    T = sh["T"] * T_scale; To = sh["T_odds"] * T_scale
    start = MA + lnorm(rng, *setup, n)

        lm = lam_m0[idx, None] * f
        cdf = np.zeros_like(lm)
        term = np.exp(-lm)
        ke = K_eff[idx, None]
        for j in range(K + 1):
            cdf += np.where(j <= ke, term, 0.0)
            term = term * lm / (j + 1)
        oc = oco0[idx, None] * 2.0 ** np.minimum(dtt / To[idx, None], 60)
        return np.exp(-ln) * np.minimum(cdf, 1) * oc / (1 + oc)

    def h(idx, t):
        return k * P(idx, t) * (t[None, :] >= start[idx, None]) * (t[None, :] >= t_tp[idx, None])

    first = grid_first(h, None, n, MA, seed=seed + 7)

    fail = sh["S0"] | (rng.random(n) < p_fail_own)
    r = summarize(eta, fail)
    if debug:

        r["lam_m0_q"] = np.round(np.quantile(lam_m0, [0.1, 0.5, 0.9]), 1).tolist()
        r["t_tp_q"] = np.round(np.quantile(t_tp, [0.1, 0.5, 0.9, 0.99]), 2).tolist() if throughput else None
        r["K_eff_share"] = {int(v): round(float((K_eff == v).mean()), 3) for v in np.unique(K_eff)}
    return r

# ---------------- Fukushima ----------------
def fukushima(n=400_000, seed=20261014, sh=None, vB=0.7, vA=0.4, kd=0.3, n_eff=(4, 2, 8),
              pval=(0.4, 0.15, 0.7), nav_mult=1.0, p_fail_own=0.03, T_scale=1.0, nav_beta=(2, 8)):
    rng = np.random.default_rng(seed)
    sh = sh or shared(np.random.default_rng(seed + 1), n)
    t0 = 2026.30
    # nav: ASSUMED per-round autonomous course-completion rate (Apr 2026). Measured: >= 1 completion in 8
    # autonomous rounds (Tien Kung); the other 7 outcomes are NOT reported (audit 2026-10-14-r2 finding 1), so
    # Beta(2,8) (mean 0.2) is an assumed prior, not a count; alternatives in the sensitivity table.
    pn = rng.beta(*nav_beta, n) * nav_mult
    on0 = pn / (1 - pn)
    ov0 = odds_lnorm(rng, *pval, n)
    ne = np.clip(np.round(lnorm(rng, *n_eff, n)), 1, 20)
    To = sh["T_odds"] * T_scale
    years = np.arange(2027, 2101)
    flB = rng.random((n, years.size)) < vB                   # Beijing E-Town (April) qualifies that year
    flA = rng.random((n, years.size)) < vA                   # autumn venue (WRS F-REI style) qualifies

    pj0 = (on0 / (1 + on0)) * (ov0 / (1 + ov0))             # joint nav+repair success per attempt, Apr 2026
    oj0 = pj0 / (1 - pj0)                                    # the Dyna driver measures whole-task failure odds,
                                                             # so the JOINT odds halve every T_odds
    def p(idx, t):
        o = oj0[idx, None] * 2.0 ** np.minimum((t[None, :] - t0) / To[idx, None], 60)
        return o / (1 + o)

    def h(idx, t):
        return kd * p(idx, t)

    def j(idx, t):
        J = np.zeros((len(idx), t.size))
        pp = p(idx, t)
        qv = 1 - (1 - pp) ** ne[idx, None]
        for yi, y in enumerate(years):
            for off, fl in ((0.30, flB), (0.80, flA)):
                ti = int(round((y + off - t[0]) / (t[1] - t[0])))
                if 0 <= ti < t.size:
                    J[:, ti] += np.where(fl[idx, yi], -np.log(np.maximum(1 - qv[:, ti], 1e-12)), 0)
        return J

    eta = grid_first(h, j, n, MA, seed=seed + 7)
    fail = sh["S0"] | (rng.random(n) < p_fail_own)
    r = summarize(eta, fail)
    r["q_venue_2026_med"] = round(float(np.median(1 - (1 - (on0 / (1 + on0)) * (ov0 / (1 + ov0))) ** ne)), 3)
    return r


# ---------------- Wozniak ----------------
def wozniak(n=400_000, seed=20261014, sh=None, p0=(0.25, 0.10, 0.45), theta=0.35, c=1.75, m=5,
            w_leap=0.25, leap_med=2033.0, p_fail_own=0.01, T_scale=1.0, debug=False, all_homes=False):
    # all_homes=True: a campaign passes only if EVERY one of its m homes succeeds (p^m) - the LEAP-style
    # three-home definition (audit 2026-10-14-r2 finding 4); False: first success in any home (base reading).
    rng = np.random.default_rng(seed)
    sh = sh or shared(np.random.default_rng(seed + 1), n)
    t0 = 2026.7
    o0 = odds_lnorm(rng, *p0, n)
    To = sh["T_odds"] * T_scale
    oth = theta / (1 - theta)
    t_th = t0 + np.maximum(0, np.log2(oth / o0)) * To       # coffee enters unseen-home evaluations

    def h(idx, t):
        o = o0[idx, None] * 2.0 ** np.minimum((t[None, :] - t0) / To[idx, None], 60)
        pp = o / (1 + o)
        ps = pp ** m if all_homes else 1 - (1 - pp) ** m
        return c * ps * (t[None, :] >= t_th[idx, None])

    trend = grid_first(h, None, n, MA, seed=seed + 7)
    # LEAP branch: 3 homes without mistakes at 50% joint = 0.794/home (odds 3.85); mapped to one success by
    # subtracting the trend's time from the inclusion level (theta) to the 0.794 level.
    gap = np.log2(3.85 / oth) * To
    # spread kept from the previous working (80% range ~ x0.52..x1.83 of the time to the centre, sigma 0.49)
    leap = 2026.2 + np.maximum(leap_med - 2026.2 - gap, 0.1) * np.exp(0.49 * rng.standard_normal(n))
    use_leap = rng.random(n) < w_leap
    eta = np.where(use_leap, leap, trend)
    fail = sh["S0"] | (rng.random(n) < p_fail_own)
    r = summarize(eta, fail)
    if debug:
        r["trend_only"] = summarize(trend, fail)
        r["leap_only"] = summarize(leap, fail)
        r["gap_med"] = round(float(np.median(gap)), 2)
        r["t_th_med"] = round(float(np.median(t_th)), 2)
    return r
